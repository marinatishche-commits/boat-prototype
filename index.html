function toggleAccordion(btn) {

    const body = btn.nextElementSibling;

    const icon = btn.querySelector('.accordion-icon');

    const isOpen = body.classList.contains('open');

 

    // Закрыть все

    document.querySelectorAll('.accordion-body').forEach(b => b.classList.remove('open'));

    document.querySelectorAll('.accordion-icon').forEach(i => i.classList.remove('open'));

 

    if (!isOpen) {

        body.classList.add('open');

        icon.classList.add('open');

    }

}

 

/* ??? CALENDAR ??? */

function buildCalendar() {

    const now   = new Date();

    const year  = now.getFullYear();

    const month = now.getMonth();

    const monthNames = [

        'Январь','Февраль','Март','Апрель','Май','Июнь',

        'Июль','Август','Сентябрь','Октябрь','Ноябрь','Декабрь'

    ];

    document.getElementById('calendar-month').textContent =

        `${monthNames[month]} ${year}`;

 

    const grid        = document.getElementById('calendar-grid');

    grid.innerHTML    = '';

    const firstDay    = new Date(year, month, 1).getDay();

    const offset      = firstDay === 0 ? 6 : firstDay - 1;

    const daysInMonth = new Date(year, month + 1, 0).getDate();

    const today       = now.getDate();

 

    for (let i = 0; i < offset; i++) {

        grid.insertAdjacentHTML('beforeend', '<div></div>');

    }

    for (let d = 1; d <= daysInMonth; d++) {

        const isPast  = d < today;

        const isToday = d === today;

        const cls = [

            'date-cell p-2 text-xs rounded-xl border border-slate-100 cursor-pointer font-medium',

            isPast  ? 'disabled' : '',

            isToday ? 'today'    : ''

        ].filter(Boolean).join(' ');

        grid.insertAdjacentHTML('beforeend',

            `<div class="${cls}" data-day="${d}" onclick="selectDate(this)">${d}</div>`

        );

    }

}

 

function selectDate(el) {

    if (el.classList.contains('disabled')) return;

    document.querySelectorAll('.date-cell').forEach(c => c.classList.remove('selected'));

    el.classList.add('selected');

    const now  = new Date();

    state.date = new Date(now.getFullYear(), now.getMonth(), parseInt(el.dataset.day));

    updateSummary();

}

function pluralHours(n) {

    if (n === 1) return 'час';

    if (n >= 2 && n <= 4) return 'часа';

    return 'часов';

}

/* ??? TOAST ??? */

function showToast(msg) {

    const t   = document.getElementById('toast');

    t.textContent  = msg;

    t.style.opacity = '1';

    clearTimeout(t._timer);

    t._timer = setTimeout(() => { t.style.opacity = '0'; }, 2500);

}

 

/* ??? PHONE MASK ??? */

document.getElementById('input-phone').addEventListener('input', function () {

    let v = this.value.replace(/\D/g, '');

    if (v.startsWith('8')) v = '7' + v.slice(1);

    if (!v) { this.value = ''; return; }

    let r = '+7';

    if (v.length > 1) r += ' (' + v.slice(1, 4);

    if (v.length >= 4) r += ') ' + v.slice(4, 7);

    if (v.length >= 7) r += '-' + v.slice(7, 9);

    if (v.length >= 9) r += '-' + v.slice(9, 11);

    this.value = r;

});
